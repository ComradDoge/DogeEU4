# Fate of the Netherlands
country_event = {
	id = ME_Burgundy_Events.14
	title = ME_Burgundy_Events.14.t
	desc = ME_Burgundy_Events.14.d
	picture = GOOD_WITH_MONARCH_eventPicture

	is_triggered_only = yes

	option = {
		name = ME_Burgundy_Events.14.a

		ai_chance = { factor = 1 }

		trigger = {
			NOT = { exists = NED }
			any_subject_country = {
				capital_scope = {
					region = low_countries_region
				}
			}
		}

		custom_tooltip = ME_BUR_FORM_NETHERLANDS_VASSAL
		hidden_effect = {
			random_subject_country = {
				limit = {
					capital_scope = {
						region = low_countries_region
					}
				}
				save_event_target_as = me_bur_netherlands_subject
				set_country_flag = me_bur_netherlands_subject_flag
			}
			every_subject_country = {
				limit = {
					capital_scope = {
						region = low_countries_region
					}
					NOT = { has_country_flag = me_bur_netherlands_subject_flag }
				}
				event_target:me_bur_netherlands_subject = {
					inherit = PREV
				}
			}
			event_target:me_bur_netherlands_subject = {
				change_tag = NED
				swap_non_generic_missions = yes
				swap_free_idea_group = yes
			}
		}
	}

	option = {
		name = ME_Burgundy_Events.14.b

		ai_chance = { factor = 1 }

		custom_tooltip = ME_BUR_ANNEX_LOW_COUNTRIES_BIG_PENALTY
		hidden_effect = {
			every_subject_country = {
				limit = {
					capital_scope = {
						region = low_countries_region
					}
				}
				every_owned_province = {
					limit = {
						region = low_countries_region
					}
					cede_province = ROOT
					add_core = ROOT
					add_unrest = 10
					add_local_autonomy = 60
				}
			}
		}
	}
}

# Shattered: setup when the Europa Expanded mission tree is picked (EE did this in history)
country_event = {
	id = ME_Burgundy_Events.22
	title = ME_Burgundy_Events.22.t
	desc = ME_Burgundy_Events.22.d
	picture = GOOD_WITH_MONARCH_eventPicture
	hidden = yes

	is_triggered_only = yes

	trigger = {
		OR = {
			tag = BUR
			tag = LOT
		}
		ME_bur_uses_ee_missions_trigger = yes
	}

	immediate = {
		# Re-granted if the tree was switched away (privilege becomes invalid) and picked again
		if = {
			limit = {
				has_estate = estate_nobles
				NOT = { has_estate_privilege = ME_BUR_order_of_the_golden_fleece }
			}
			set_estate_privilege = ME_BUR_order_of_the_golden_fleece
		}
		# Only the first time
		if = {
			limit = {
				NOT = { has_country_flag = me_bur_tree_setup_done_flag }
			}
			set_country_flag = me_bur_tree_setup_done_flag
			if = {
				limit = {
					has_government_mechanic = golden_fleece_mechanic
				}
				set_government_power = {
					mechanic_type = golden_fleece_mechanic
					power_type = golden_fleece_support
					value = 20
				}
			}
		}
		calculate_council_support = yes
	}

	option = {
		name = ME_Burgundy_Events.22.a
	}
}
