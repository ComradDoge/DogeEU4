# The Avignon Question (occ_avignon_question)
country_event = {
	id = occ_events.4
	title = occ_events.4.t
	desc = occ_events.4.d
	picture = POPE_PREACHING_eventPicture
	goto = 202

	is_triggered_only = yes

	option = { # A Cathar seat
		name = occ_events.4.a
		trigger = {
			religion = cathar
			202 = { owned_by = ROOT }
		}
		occ_change_paratge = { value = -15 }
		202 = {
			change_religion = cathar
			add_reform_center = cathar
		}
		ai_chance = { factor = 40 }
	}
	option = { # Spare the palace
		name = occ_events.4.b
		occ_change_paratge = { value = 5 }
		add_prestige_or_monarch_power = { amount = 25 }
		202 = {
			if = {
				limit = { owned_by = ROOT }
				add_base_tax = 1
			}
		}
		ai_chance = { factor = 40 }
	}
	option = { # Sell it back
		name = occ_events.4.c
		add_years_of_income = 1
		ai_chance = { factor = 20 }
	}
}

# The Throne of Peter (occ_throne_of_peter)
# c needs Paratge 50+, d needs Paratge below -50 (Zeal)
country_event = {
	id = occ_events.6
	title = occ_events.6.t
	desc = occ_events.6.d
	picture = RELIGIOUS_CONVERSION_eventPicture
	goto = 118
	major = yes

	is_triggered_only = yes

	option = { # Burn the archives
		name = occ_events.6.a
		occ_change_paratge = { value = -20 }
		add_country_modifier = {
			name = occ_burned_curia_modifier
			duration = 9125
		}
		118 = {
			add_province_modifier = {
				name = occ_ashes_of_the_curia_modifier
				duration = 3650
			}
		}
		custom_tooltip = occ_burned_curia_opinion_tt
		hidden_effect = {
			every_country = {
				limit = {
					religion = catholic
				}
				add_opinion = {
					who = ROOT
					modifier = occ_burned_the_curia_opinion
				}
			}
		}
		ai_chance = {
			factor = 30
			modifier = {
				factor = 2
				NOT = {
					has_government_power = {
						mechanic_type = occ_paratge_mechanic
						power_type = paratge
						value = 0
					}
				}
			}
		}
	}
	option = { # Keep them
		name = occ_events.6.b
		occ_change_paratge = { value = 10 }
		add_prestige_or_monarch_power = { amount = 50 }
		118 = {
			add_province_modifier = {
				name = occ_archive_of_sentences_modifier
				duration = -1
			}
		}
		ai_chance = { factor = 30 }
	}
	option = { # Safe conduct for the cardinals
		name = occ_events.6.c
		trigger = {
			has_government_power = {
				mechanic_type = occ_paratge_mechanic
				power_type = paratge
				value = 50
			}
		}
		occ_change_paratge = { value = 15 }
		add_country_modifier = {
			name = occ_safe_conduct_modifier
			duration = 9125
		}
		custom_tooltip = occ_spared_cardinals_opinion_tt
		hidden_effect = {
			every_country = {
				limit = {
					religion = catholic
				}
				add_opinion = {
					who = ROOT
					modifier = occ_spared_the_cardinals_opinion
				}
			}
		}
		ai_chance = { factor = 60 }
	}
	option = { # Counter-crusade
		name = occ_events.6.e
		trigger = {
			NOT = {
				has_government_power = {
					mechanic_type = occ_paratge_mechanic
					power_type = paratge
					value = -50
				}
			}
		}
		occ_change_paratge = { value = -15 }
		add_country_modifier = {
			name = occ_counter_crusade_modifier
			duration = 9125
		}
		ai_chance = { factor = 60 }
	}
}
