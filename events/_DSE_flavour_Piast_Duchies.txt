# A Piast Match (once per duchy)
country_event = {
	id = piast_events.8
	title = piast_events.8.t
	desc = piast_events.8.d
	picture = DIPLOMACY_SIGNING_eventPicture

	trigger = {
		is_piast_duchy_trigger = yes
		government = monarchy
		NOT = { has_country_flag = piast_match_offered }
		is_at_war = no
		any_country = {
			is_piast_duchy_trigger = yes
			NOT = { tag = ROOT }
			government = monarchy
			NOT = { war_with = ROOT }
			NOT = { marriage_with = ROOT }
		}
	}

	mean_time_to_happen = {
		months = 120
	}

	immediate = {
		hidden_effect = {
			set_country_flag = piast_match_offered
			random_country = {
				limit = {
					is_piast_duchy_trigger = yes
					NOT = { tag = ROOT }
					government = monarchy
					NOT = { war_with = ROOT }
					NOT = { marriage_with = ROOT }
				}
				save_event_target_as = piast_match_target
			}
		}
	}

	option = { # Accept the match
		name = piast_events.8.a
		event_target:piast_match_target = {
			create_marriage = ROOT
			add_opinion = {
				who = ROOT
				modifier = piast_marriage_bond
			}
			reverse_add_opinion = {
				who = ROOT
				modifier = piast_marriage_bond
			}
		}
		ai_chance = { factor = 60 }
	}
	option = { # Refuse
		name = piast_events.8.b
		add_prestige_or_monarch_power = { amount = 5 }
		event_target:piast_match_target = {
			add_opinion = {
				who = ROOT
				modifier = piast_match_refused
			}
		}
		ai_chance = { factor = 40 }
	}
}
