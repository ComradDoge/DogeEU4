# The Homage of Qurtuba (sev_qurtuba_the_old_capital): Sevilla asks the other taifas to bow. FROM is Sevilla.
country_event = {
	id = taifa_events.3
	title = taifa_events.3.t
	desc = taifa_events.3.d
	picture = DIPLOMACY_SIGNING_eventPicture

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			clr_country_flag = taifa_paid_homage
			clr_country_flag = taifa_refused_homage
		}
	}

	option = { # Bow to Qurtuba
		name = taifa_events.3.a
		trigger = {
			is_subject = no
			NOT = { war_with = FROM }
		}
		FROM = {
			vassalize = ROOT
		}
		add_opinion = {
			who = FROM
			modifier = taifa_homage_paid
		}
		hidden_effect = {
			set_country_flag = taifa_paid_homage
			FROM = {
				country_event = { id = taifa_events.4 }
			}
		}
		ai_chance = {
			factor = 10
			modifier = {
				factor = 6
				FROM = {
					army_strength = {
						who = ROOT
						value = 2
					}
				}
			}
			modifier = {
				factor = 2
				is_at_war = yes
			}
		}
	}
	option = { # We bow to no one
		name = taifa_events.3.b
		capital_scope = {
			add_permanent_claim = FROM
		}
		add_opinion = {
			who = FROM
			modifier = taifa_homage_refused
		}
		reverse_add_opinion = {
			who = FROM
			modifier = taifa_homage_refused
		}
		add_prestige_or_monarch_power = { amount = 10 }
		hidden_effect = {
			set_country_flag = taifa_refused_homage
			FROM = {
				country_event = { id = taifa_events.4 }
			}
		}
		ai_chance = { factor = 90 }
	}
}
