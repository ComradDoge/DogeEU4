# The Piotrkow Offer (maz_the_piotrkow_offer). Humans reach it through Select Current Mission Branch
# (EE_select_current_branch) with only the previewed option; the AI gets it straight from the mission.
country_event = {
	id = maz_events.1
	title = maz_events.1.t
	desc = maz_events.1.d
	picture = DIPLOMACY_SIGNING_eventPicture
	goto = capital
	major = yes

	is_triggered_only = yes

	option = { # Kneel to the Jagiellon
		name = maz_events.1.a
		trigger = {
			NOT = { has_country_flag = maz_preview_candidacy }
		}
		custom_tooltip = maz_crown_fealty_tt
		custom_tooltip = maz_crown_no_annex_tt
		hidden_effect = {
			set_country_flag = maz_crown_fealty
			clr_country_flag = maz_preview_fealty
			change_variable = { # creates the blood counter at 0 so its tooltips have a value (an unset variable logs "Unknown text property")
				which = maz_crown_blood
				value = 0
			}
			swap_non_generic_missions = yes
		}
		# Humans only (like the annexation block): the AI stays an ordinary vassal that leaves the fighting to Poland
		if = {
			limit = {
				ai = no
				is_subject_of = POL
			}
			custom_tooltip = maz_sworn_duchy_tt
			hidden_effect = {
				POL = {
					create_subject = {
						subject_type = maz_sworn_duchy
						subject = ROOT
						keep_liberty_desire = yes
						message = no
					}
				}
			}
		}
		add_legitimacy_or_mil_power = { amount = 10 republican_tradition = 5 }
		add_liberty_desire = -25
		ai_chance = {
			factor = 60
			modifier = {
				factor = 0.25
				liberty_desire = 50
			}
		}
	}
	option = { # Take the lords at their word
		name = maz_events.1.b
		trigger = {
			NOT = { has_country_flag = maz_preview_fealty }
		}
		custom_tooltip = maz_crown_candidacy_tt
		hidden_effect = {
			set_country_flag = maz_crown_candidacy
			clr_country_flag = maz_preview_candidacy
			swap_non_generic_missions = yes
		}
		add_prestige_or_monarch_power = { amount = 15 }
		add_liberty_desire = 15
		if = {
			limit = {
				exists = POL
			}
			POL = {
				add_opinion = {
					who = ROOT
					modifier = maz_rival_for_the_crown
				}
			}
		}
		ai_chance = {
			factor = 40
			modifier = {
				factor = 2
				liberty_desire = 50
			}
		}
	}
}

# The Crown Treasurer's clerk checks our muster rolls. Sent by maz_send_clerk_effect after a forged roll
# (Mazovia options in events/ArmyProfessionalism.txt, common/scripted_effects/DST_Mazovia_scripted_effects.txt)
country_event = {
	id = maz_events.10
	title = maz_events.10.t
	desc = maz_events.10.d
	picture = ADVISOR_eventPicture

	is_triggered_only = yes

	trigger = {
		maz_can_forge_rolls_trigger = yes
		check_variable = {
			which = maz_forged_rolls
			value = 1
		}
	}

	immediate = {
		hidden_effect = {
			clr_country_flag = maz_clerk_coming
		}
	}

	option = { # Pay him to lose the ledger: the forged count stays, so the next clerk is as likely
		name = maz_events.10.a
		add_years_of_income = -0.5
		custom_tooltip = maz_clerk_bribe_tt
		ai_chance = {
			factor = 50
		}
	}
	option = { # Hang a quartermaster: the forged blood is struck off, not below 0
		name = maz_events.10.b
		custom_tooltip = maz_clerk_hang_tt
		hidden_effect = {
			if = {
				limit = {
					check_variable = {
						which = maz_crown_blood
						which = maz_forged_blood
					}
				}
				subtract_variable = {
					which = maz_crown_blood
					which = maz_forged_blood
				}
			}
			else = {
				set_variable = {
					which = maz_crown_blood
					value = 0
				}
			}
			maz_forged_rolls_reset_effect = yes
			if = {
				limit = { has_country_flag = maz_fiefs_unlocked }
				maz_fief_prices_refresh_effect = yes
			}
		}
		ai_chance = {
			factor = 40
		}
	}
	option = { # Swear to it on our oath: the blood stays, Poland trusts us less
		name = maz_events.10.c
		trigger = {
			POL = {
				has_opinion = {
					who = ROOT
					value = 100
				}
			}
		}
		POL = {
			add_opinion = {
				who = ROOT
				modifier = maz_false_muster_rolls
			}
		}
		custom_tooltip = maz_clerk_sworn_tt
		hidden_effect = {
			maz_forged_rolls_reset_effect = yes
		}
		ai_chance = {
			factor = 10
		}
	}
}

#### 5-year pulse events about blood for the Crown (on_five_year_pulse_4 in common/on_actions/DST_Mazovia_on_actions.txt).
# A sworn Mazovia only, each at most once every 30 years (maz_pulse_event_ready_trigger). Options that pad the blood go through

# Widows at Krakow: men entered as fallen are alive at home, and their wives claim the dead men's wages. Needs a forged roll
country_event = {
	id = maz_events.11
	title = maz_events.11.t
	desc = maz_events.11.d
	picture = MARCHING_PEASANTS_eventPicture

	is_triggered_only = yes

	trigger = {
		maz_pulse_event_ready_trigger = { flag = maz_widows_at_krakow }
		check_variable = {
			which = maz_forged_rolls
			value = 1
		}
	}

	immediate = {
		hidden_effect = {
			clr_country_flag = maz_widows_at_krakow
			set_country_flag = maz_widows_at_krakow
		}
	}

	option = { # Pay them ourselves: costs money, adds no risk
		name = maz_events.11.a
		add_years_of_income = -0.5
		custom_tooltip = maz_widows_paid_tt
		ai_chance = {
			factor = 50
		}
	}
	option = { # Let Krakow pay: more forged blood
		name = maz_events.11.b
		maz_forge_rolls_effect = { blood = 500 rolls = 1 }
		ai_chance = {
			factor = 20
		}
	}
	option = { # Strike the husbands from the rolls: lose blood, not below 0, the forged count starts over
		name = maz_events.11.c
		custom_tooltip = maz_widows_struck_tt
		hidden_effect = {
			if = {
				limit = {
					check_variable = {
						which = maz_crown_blood
						value = 1000
					}
				}
				subtract_variable = {
					which = maz_crown_blood
					value = 1000
				}
			}
			else = {
				set_variable = {
					which = maz_crown_blood
					value = 0
				}
			}
			maz_forged_rolls_reset_effect = yes
			if = {
				limit = { has_country_flag = maz_fiefs_unlocked }
				maz_fief_prices_refresh_effect = yes
			}
		}
		ai_chance = {
			factor = 30
		}
	}
}

# Blood for the Crown, battle settlement (maz_add_crown_blood_effect schedules it the day after a battle):
# our share of the side's losses = pending x our regiments / allied regiments in the battle province (maz_crown_blood_battle_share_effect)
country_event = {
	id = maz_events.98
	title = maz_events.98.t
	desc = maz_events.98.d
	picture = BATTLE_eventPicture
	hidden = yes

	is_triggered_only = yes

	immediate = {
		log = "MAZ crown blood: side lost [Root.maz_pending_blood.GetValue], our regiments [Root.maz_battle_own.GetValue] of [Root.maz_battle_side.GetValue] on our side"
		# Blood only counts from battles with at least 4 Mazovian regiments (no blood from token stacks)
		if = {
			limit = {
				check_variable = {
					which = maz_battle_own
					value = 4
				}
			}
			multiply_variable = {
				which = maz_pending_blood
				which = maz_battle_own
			}
			divide_variable = {
				which = maz_pending_blood
				which = maz_battle_side
			}
		}
		else = {
			set_variable = {
				which = maz_pending_blood
				value = 0
			}
		}
		change_variable = {
			which = maz_crown_blood
			which = maz_pending_blood
		}
		log = "MAZ crown blood: [Root.GetName] adds [Root.maz_pending_blood.GetValue], total now [Root.maz_crown_blood.GetValue]"
		set_variable = {
			which = maz_pending_blood
			value = 0
		}
		set_variable = {
			which = maz_battle_own
			value = 0
		}
		set_variable = {
			which = maz_battle_side
			value = 0
		}
		clr_country_flag = maz_blood_settling
		if = {
			limit = { has_country_flag = maz_fiefs_unlocked }
			maz_fief_prices_refresh_effect = yes
		}
	}

	option = {
		name = maz_events.98.a
	}
}
