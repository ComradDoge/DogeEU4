		gv_is_galicia_volhynia_trigger = yes

		is_subject = no
		is_at_war = no

			gv_is_galicia_volhynia_trigger = yes

			is_subject = no
			NOT = { war_with = ROOT }
			NOT = { alliance_with = ROOT }
			NOT = { has_country_flag = gv_accord_offered }
		}
	}

	mean_time_to_happen = {
		months = 60
	}

	immediate = {
		hidden_effect = {
			set_country_flag = gv_accord_offered

				limit = {
					gv_is_galicia_volhynia_trigger = yes
					NOT = { tag = ROOT }

					NOT = { alliance_with = ROOT }
