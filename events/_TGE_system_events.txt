# Recalculation. Fired by tge_request_recalc (debounced, with a cooldown), and directly by the
# resources button and decisions.
country_event = {
	id = New_TradeGood_Events.2
	title = New_TradeGood_Events.2.t
	desc = New_TradeGood_Events.2.d
	picture = TRADEGOODS_eventPicture

	is_triggered_only = yes
	hidden = yes

	immediate = {
		hidden_effect = {
			# Fail-safe: at most 10 queued AI recalculations a day world-wide; the rest move to one of the next days
			if = {
				limit = {
					ai = yes
					has_country_flag = tge_recalc_pending
					tge_recalc_budget_full = yes
				}
				log = "TGE RECALC deferred [Root.GetTag]"
				country_event = { id = New_TradeGood_Events.2 days = 1 random = 6 }	# tge_recalc_pending stays set, so no other request queues
			}
			else = {
				if = {
					limit = {
						ai = yes
						has_country_flag = tge_recalc_pending
					}
					tge_recalc_take_slot = yes
				}
				clr_country_flag = tge_recalc_pending
				log = "TGE RECALC [Root.GetTag]"	# game.log: how often each country recalculates
				tge_recalc_and_notify = yes
			}
		}
	}

	option = {
		name = New_TradeGood_Events.2.a
		ai_chance = { factor = 100 }
	}
}
