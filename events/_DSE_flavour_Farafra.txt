# Where Shall the Throne Sit? (faf_the_city_of_the_sultans)
country_event = {
	id = faf_events.2
	title = faf_events.2.t
	desc = faf_events.2.d
	picture = Cairo_eventPicture
	goto = faf_cairo

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			361 = { save_event_target_as = faf_cairo }
		}
	}

	option = { # Rule from Cairo
		name = faf_events.2.a
		trigger = {
			owns = 361
			NOT = { capital = 361 }
		}
		361 = { move_capital_effect = yes }
		add_prestige_or_monarch_power = { amount = 25 }
		ai_chance = { factor = 70 }
	}
	option = { # The throne stays where it is
		name = faf_events.2.b
		capital_scope = {
			if = {
				limit = { area = al_wahat_area }
				add_province_modifier = {
					name = faf_the_oasis_throne_modifier
					duration = -1
				}
			}
		}
		add_adm_power = 50
		ai_chance = { factor = 30 }
	}
}
