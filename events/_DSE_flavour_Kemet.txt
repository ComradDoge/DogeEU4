# The Two Lands United (formation of Kemet)
country_event = {
	id = egp_events.1
	title = egp_events.1.t
	desc = egp_events.1.d
	picture = EgyptianSetHorus_eventPicture
	major = yes

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			361 = { save_event_target_as = egp_menefer }
			360 = { save_event_target_as = egp_waset }
		}
	}

	option = { # Rule from Men-nefer
		name = egp_events.1.a
		trigger = {
			owns = 361
			NOT = { capital = 361 }
		}
		361 = { move_capital_effect = yes }
		add_prestige_or_monarch_power = { amount = 25 }
		ai_chance = { factor = 60 }
	}
	option = { # Rule from Waset
		name = egp_events.1.b
		trigger = {
			owns = 360
			NOT = { capital = 360 }
		}
		360 = {
			move_capital_effect = yes
			add_base_tax = 1
			add_province_modifier = {
				name = egp_the_throne_of_waset_modifier
				duration = -1
			}
		}
		add_estate_church_loyalty_effect = yes
		ai_chance = { factor = 20 }
	}
	option = { # The throne stays where it is
		name = egp_events.1.c
		add_stability_or_adm_power = yes
		ai_chance = { factor = 20 }
	}
}

# The Expedition to Punt (egp_the_expedition_to_punt)
country_event = {
	id = egp_events.4
	title = egp_events.4.t
	desc = egp_events.4.d
	picture = EXPLORERS_eventPicture
	goto = 2321

	is_triggered_only = yes

	option = { # The incense trees, for the gods
		name = egp_events.4.a
		360 = {
			if = {
				limit = {
					owned_by = ROOT
					NOT = { trade_goods = incense }
				}
				change_trade_goods = incense
			}
		}
		add_estate_church_loyalty_effect = yes
		ai_chance = { factor = 30 }
	}
	option = { # Gold, ebony and ivory, for the treasury
		name = egp_events.4.b
		add_years_of_income = 1
		ai_chance = { factor = 40 }
	}
	option = { # A pact with the chiefs of Punt
		name = egp_events.4.c
		add_country_modifier = {
			name = egp_the_road_to_punt_modifier
			duration = 9125
		}
		ai_chance = { factor = 30 }
	}
}
