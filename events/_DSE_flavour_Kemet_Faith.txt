# The Inundation
country_event = {
	id = kemet_events.20
	title = kemet_events.20.t
	desc = kemet_events.20.d
	picture = FAMINE_eventPicture
	goto = kemet_flood_province

	trigger = {
		OR = {
			NOT = { has_country_flag = kemet_nile_flood_cd }
			had_country_flag = { flag = kemet_nile_flood_cd days = 3650 }
		}
		any_owned_province = {
			region = egypt_region
			OR = {
				trade_goods = grain
				trade_goods = sugar
				trade_goods = cotton
				trade_goods = dates
				trade_goods = paper
			}
			NOT = { has_province_modifier = kemet_rich_flood_modifier }
			NOT = { has_province_modifier = kemet_flood_damage_modifier }
		}
	}

	mean_time_to_happen = {
		years = 12
	}

	immediate = {
		hidden_effect = {
			clr_country_flag = kemet_nile_flood_cd
			set_country_flag = kemet_nile_flood_cd
			random_owned_province = {
				limit = {
					region = egypt_region
					OR = {
						trade_goods = grain
						trade_goods = sugar
						trade_goods = cotton
						trade_goods = dates
						trade_goods = paper
					}
					NOT = { has_province_modifier = kemet_rich_flood_modifier }
					NOT = { has_province_modifier = kemet_flood_damage_modifier }
				}
				save_event_target_as = kemet_flood_province
			}
		}
	}

	option = { # The surplus goes to the granaries of the state
		name = kemet_events.20.a
		add_years_of_income = 0.15
		ai_chance = { factor = 50 }
	}
	option = { # Leave it with the farmers
		name = kemet_events.20.b
		event_target:kemet_flood_province = {
			add_prosperity = 25
			add_province_modifier = {
				name = kemet_rich_flood_modifier
				duration = 1825
			}
		}
		ai_chance = { factor = 50 }
	}
}
