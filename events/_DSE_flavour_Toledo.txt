# Raiders from the South (MTTH): the border with the Muslim south
country_event = {
	id = tol_events.11
	title = tol_events.11.t
	desc = tol_events.11.d
	picture = VILLAGE_RAIDED_eventPicture
	goto = capital

	trigger = {
		tag = TOL
		is_at_war = no
		any_neighbor_country = {
			religion_group = muslim
			capital_scope = {
				region = iberia_region
			}
		}
		OR = {
			NOT = { has_country_flag = tol_raiders_flag }
			had_country_flag = {
				flag = tol_raiders_flag
				days = 3650
			}
		}
	}

	mean_time_to_happen = {
		months = 120
	}

	immediate = {
		hidden_effect = {
			clr_country_flag = tol_raiders_flag
			set_country_flag = tol_raiders_flag
		}
	}

	option = { # Ride after them
		name = tol_events.11.a
		add_army_tradition = 10
		random_owned_province = {
			limit = {
				any_neighbor_province = {
					owner = {
