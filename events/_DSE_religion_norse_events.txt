	picture = CAROLEANS_eventPicture
	goto = dse_norse_einherjar_muster

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			if = {
				limit = {
					any_owned_province = { has_province_flag = dse_norse_einherjar_muster_flag }
				}
				random_owned_province = {
					limit = { has_province_flag = dse_norse_einherjar_muster_flag }
					save_event_target_as = dse_norse_einherjar_muster
				}
			}
			else = {
				capital_scope = { save_event_target_as = dse_norse_einherjar_muster }
			}
			every_owned_province = {
				limit = { has_province_flag = dse_norse_einherjar_muster_flag }
				clr_province_flag = dse_norse_einherjar_muster_flag
			}
		}
	}

	option = {		# Raise a rune stone on the muster field
		name = dse_norse_religious_events.11.a
		ai_chance = { factor = 2 }
		event_target:dse_norse_einherjar_muster = {
			add_province_modifier = {
				name = dse_norse_einherjar_rune_stone_modifier
				duration = -1
			}
		}
	}
	option = {		# Feast them as heroes
		name = dse_norse_religious_events.11.b
		ai_chance = { factor = 1 }
		add_army_tradition = 10
		add_prestige_or_monarch_power = { amount = 10 }

		name = dse_norse_religious_events.11.c
		trigger = { has_personal_deity = odin }

		}
