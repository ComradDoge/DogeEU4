namespace = ai_building_events


# The Building AI is somewhat limited and doesn't queue building like a player would. 
# This is a way to save some of that building time and decision making.
# They don't cheat with cost here, but their time is reduced. 
# However, to trigger the events at all, the amount of cash they need to hold is still static and does not take in account any reductions. So it kinda works out.

# Cotton/Cloth > Cloth Workshop
# Cloth Workshop > Uniform Workshop
# Marble/Jade > Sculptures
# Lumber > Woodworking (tropical wood left alone)
# Livestock > 95% Leather 5% Cheese (hehe)
# Amber/Gems/Pearls to Jewellery but to a limit of 4
# Fur > Fur Clothing
# Soldier Households
# Road Buildings
# Mines (when short of a factory resource)
# Resource factories (cannon, metalworking, munitions, steam engine line, spectacles)

# Cotton/Cloth > Cloth Workshop
country_event = { 
	id = ai_building_events.1
	title = ai_building_events.1.t
	desc = ai_building_events.1.d
	picture = SCOTTISH_MERCENARIES_eventPicture
	
	hidden = yes

	mean_time_to_happen = {
		months = 6
		

		modifier = {
			factor = 0.5
			treasury = 5000
		}
		modifier = {
			factor = 0.8
			NOT = { num_of_loans = 1 }
		}
		modifier = {
			factor = 6
			num_of_loans = 8
		}
		modifier = {
			factor = 3
			num_of_loans = 4
		}
		modifier = {
			factor = 1.5
			num_of_loans = 2
		}
	}
	
	trigger = {
        ai = yes
        NOT = { government = native }
        #was_player = no
        treasury = 400
        OR = {
            adm_tech = 4
            overlord = { adm_tech = 4 }
        }
        any_owned_province = {
            development = 10
            meet_basic_province_requirements_trigger = yes
            no_ongoing_construction_triggers = yes
            OR = { 
                trade_goods = cotton
                trade_goods = wool
            }
        }
	}

    immediate = {
        random_owned_province = {
            limit = {
                development = 10
                meet_basic_province_requirements_trigger = yes
                no_ongoing_construction_triggers = yes
                NOT = { has_building = cloth_workshop }
                OR = { 
                    trade_goods = cotton
                    trade_goods = wool
                }
            }
            add_building_construction = {
                building = cloth_workshop
                speed = 0.5
                cost = 1
            }
        }        
    }

	option = {
		name = ai_building_events.1.main
	}
}

# Road Buildings
# Queues the best road the country's tech allows, bigger roads first, and more of them when the treasury is large.
country_event = { 
	id = ai_building_events.10
	title = ai_building_events.10.t
	desc = ai_building_events.10.d
	picture = SCOTTISH_MERCENARIES_eventPicture
	
	hidden = yes

	mean_time_to_happen = {
		months = 12

		modifier = {
			factor = 0.5
			treasury = 5000
		}
		modifier = {
			factor = 0.5
			treasury = 20000
		}
		modifier = {
			factor = 0.8
			NOT = { num_of_loans = 1 }
		}
		modifier = {
			factor = 6
			num_of_loans = 8
		}
		modifier = {
			factor = 3
			num_of_loans = 4
		}
		modifier = {
			factor = 1.5
			num_of_loans = 2
		}
	}
	
	trigger = {
		ai = yes
		NOT = { government = native }
		OR = { 
			AND = { 
				treasury = 500
				NOT = { adm_tech = 27 }
			}
			AND = { 
				treasury = 1000
				adm_tech = 27
			}
		}
		any_owned_province = { ai_road_site_any_trigger = yes }
	}

	immediate = {
		if = {
			limit = { any_owned_province = { ai_road_site_any_trigger = yes } }
			ai_build_best_road_effect = yes
		}
		else = {
			log = "ROAD BUILDER MIGHT BE LOOPING"
		}
		if = {
			limit = { treasury = 2500 }
			ai_build_best_road_effect = yes
		}
		if = {
			limit = { treasury = 10000 }
			ai_build_best_road_effect = yes
			ai_build_best_road_effect = yes
		}
		if = {
			limit = { treasury = 50000 }
			ai_build_best_road_effect = yes
			ai_build_best_road_effect = yes
			ai_build_best_road_effect = yes
			ai_build_best_road_effect = yes
		}
		every_owned_province = {
			limit = { has_province_flag = ai_road_queued }
			clr_province_flag = ai_road_queued
		}
	}

	option = {
		name = ai_building_events.1.main
	}
}


# Mines (when short of a factory resource)
# The engine AI never builds mines (ai_will_do 0), so resource-poor countries could never grow into the factory chain.
# The dig events (_TGE_Mine_events / _TGE_mine_events) already handle the AI.
country_event = {
	id = ai_building_events.11
	title = ai_building_events.11.t
	desc = ai_building_events.11.d
	picture = SCOTTISH_MERCENARIES_eventPicture

	hidden = yes

	mean_time_to_happen = {
		months = 60

		modifier = {
			factor = 0.75
			mil_tech = 16
		}
		modifier = {
			factor = 0.5
			treasury = 5000
		}
		modifier = {
			factor = 0.8
			NOT = { num_of_loans = 1 }
		}
		modifier = {
			factor = 3
			num_of_loans = 2
		}
	}

	trigger = {
        ai = yes
        NOT = { government = native }
        is_at_war = no
        treasury = 2500
        mil_tech = 8
        NOT = { num_of_loans = 4 }
        # Short of at least one factory resource (number of provinces producing it)
        OR = {
            NOT = { copper = 1 }
            NOT = { tin = 1 }
            NOT = { iron = 1 }
            NOT = { lead = 1 }
            NOT = { zinc = 1 }
        }
        NOT = {
            any_owned_province = {
                OR = {
                    has_construction = mine
                    has_building = mine
                }
            }
        }
        any_owned_province = {
            tge_ai_mine_site = yes
        }
	}

    immediate = {
        # More production development means more output once the mine strikes
        if = {
            limit = {
                any_owned_province = {
                    tge_ai_mine_site = yes
                    base_production = 6
                }
            }
            random_owned_province = {
                limit = {
                    tge_ai_mine_site = yes
                    base_production = 6
                }
                add_building_construction = {
                    building = mine
                    speed = 0.5
                    cost = 1
                }
            }
        }
        else = {
            random_owned_province = {
                limit = {
                    tge_ai_mine_site = yes
                }
                add_building_construction = {
                    building = mine
                    speed = 0.5
                    cost = 1
                }
            }
        }
    }

	option = {
		name = ai_building_events.1.main
	}
}

# Resource factories (cannon foundry, metalworking, munitions, steam engine line, spectacles)
# The engine AI barely builds them: their only modifier is a building slot, the real benefit is the
# scripted trade good change, which it does not value. Industrial factories are left to the engine,
# which builds them. At most one of each type per firing; every type has its own one-at-a-time lock.
country_event = {
	id = ai_building_events.12
	title = ai_building_events.12.t
	desc = ai_building_events.12.d
	picture = SCOTTISH_MERCENARIES_eventPicture

	hidden = yes

	mean_time_to_happen = {
		months = 12

		modifier = {
			factor = 0.5
			treasury = 5000
		}
		modifier = {
			factor = 0.8
			NOT = { num_of_loans = 1 }
		}
		modifier = {
			factor = 1.5
			num_of_loans = 1
		}
	}

	trigger = {
        ai = yes
        NOT = { government = native }
        NOT = { num_of_loans = 2 }
        treasury = 1200
        OR = {
            AND = {
                OR = {
                    has_institution = enlightenment
                    has_global_flag = first_steam_engine_manu_already_built
                }
                tge_ai_can_queue_steam_engine_manu = yes
            }
            tge_ai_can_queue_metalworking_workshop = yes
            tge_ai_can_queue_cannon_foundry = yes
            tge_ai_can_queue_munition_factory = yes
            tge_ai_can_queue_spectacle_factory = yes
        }
	}

    immediate = {
        # Most valuable first: steam engines feed industrial factories, metalworking helps its neighbours
        if = {
            limit = {
                treasury = 4000
                OR = {
                    has_institution = enlightenment
                    has_global_flag = first_steam_engine_manu_already_built
                }
