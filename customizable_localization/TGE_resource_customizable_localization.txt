# Strategic Resources overview (dst_goods_resources_gui_tt and tge_resource_goods_desc in localisation/TGE_InfoPanel_l_english.yml).
# Reads variables and flags set by the recalculation in common/scripted_effects/TGE_resource_system.txt.
# Triggers in each defined_text are mutually exclusive.

defined_text = {
	name = GetTGEFactorySummary
	random = no
	text = {
		localisation_key = tge_res_summary_stopped
		trigger = {
			check_variable = { which = tge_factories_stopped value = 1 }
		}
	}
	text = {
		localisation_key = tge_res_summary_running
		trigger = {
			NOT = { check_variable = { which = tge_factories_stopped value = 1 } }
			check_variable = { which = tge_factories_running value = 1 }
		}
	}
	text = {
		localisation_key = tge_res_empty
		trigger = {
			NOT = { check_variable = { which = tge_factories_stopped value = 1 } }
			NOT = { check_variable = { which = tge_factories_running value = 1 } }
		}
	}
}

defined_text = {
	name = GetTGEResNone
	random = no
	text = {
		localisation_key = tge_res_none
		trigger = {
			NOT = { has_country_flag = tge_has_resource_data }
		}
	}
	text = {
		localisation_key = tge_res_empty
		trigger = {
			has_country_flag = tge_has_resource_data
		}
	}
}

defined_text = {
	name = GetTGESubWord
	random = no

	}
}

# Copper
defined_text = {
	name = GetTGERes_copper
	random = no
	text = {
		localisation_key = tge_res_row_none_copper	# short and nothing at all
		trigger = {
			has_country_flag = tge_short_copper
			NOT = { check_variable = { which = copper_total value = 0.001 } }
			NOT = { check_variable = { which = copper_used value = 0.001 } }
			NOT = { check_variable = { which = copper_reserved value = 0.001 } }
			NOT = { check_variable = { which = from_subjects_copper value = 0.001 } }
		}
	}
	text = {
		localisation_key = tge_res_row_copper
		trigger = {
			OR = {
				check_variable = { which = copper_total value = 0.001 }
				has_country_flag = tge_short_copper	# also when there is none: a stopped factory waits for it
				check_variable = { which = copper_used value = 0.001 }
				check_variable = { which = copper_reserved value = 0.001 }
				check_variable = { which = from_subjects_copper value = 0.001 }
			}
			NOT = {
				AND = {
					has_country_flag = tge_short_copper
					NOT = { check_variable = { which = copper_total value = 0.001 } }
					NOT = { check_variable = { which = copper_used value = 0.001 } }
					NOT = { check_variable = { which = copper_reserved value = 0.001 } }
					NOT = { check_variable = { which = from_subjects_copper value = 0.001 } }
				}
			}
		}
	}
	text = {
		localisation_key = tge_res_empty
		trigger = {
			NOT = {
				OR = {
					check_variable = { which = copper_total value = 0.001 }
					has_country_flag = tge_short_copper	# also when there is none: a stopped factory waits for it
					check_variable = { which = copper_used value = 0.001 }
					check_variable = { which = copper_reserved value = 0.001 }
					check_variable = { which = from_subjects_copper value = 0.001 }
				}
			}
		}
	}
}
