# Occitania - Paratge
# Attached by the occ_paratge_monarchy reform (common/government_reforms/DST_Occitania_reforms.txt)
# Two sided like parliament_vs_monarchy: starts at 0, 100 = Paratge (honour, tolerance), -100 = Zeal of the Perfecti
# No scripted triggers in here, mechanics load before them
# Faction drift lives here, factions load before mechanics so monthly_paratge is unknown there

occ_paratge_mechanic = {
	alert_icon_gfx = GFX_alerticons_government_mechanics
	alert_icon_index = 3
	available = {
		always = yes
	}

	powers = {
		paratge = {
			show_before_interactions = yes
			gui = occ_paratge_gov_mech
			max = 100	# Paratge
			min = -100	# Zeal
			base_monthly_growth = 0

			# Paratge side
			scaled_modifier = {
				trigger = {
					has_government_power = {
						mechanic_type = occ_paratge_mechanic
						power_type = paratge
						value = 0
					}
				}
				modifier = {
					trade_efficiency = 0.1
					tolerance_heretic = 2
					diplomatic_reputation = 1
				}
				start_value = 0
				end_value = 100
			}
			scaled_modifier = {
				trigger = {
					has_country_flag = occ_paratge_parliament_flag
					has_government_power = {
						mechanic_type = occ_paratge_mechanic
						power_type = paratge
						value = 0
					}
				}
				modifier = {
					global_unrest = -1
					global_tax_modifier = 0.1
				}
				start_value = 0
				end_value = 100
			}

			# Zeal side
			reverse_scaled_modifier = {
				trigger = {
					NOT = {
						has_government_power = {
							mechanic_type = occ_paratge_mechanic
							power_type = paratge
							value = 0
						}
					}
				}
				modifier = {
					tolerance_own = 2
					global_missionary_strength = 0.03
					warscore_cost_vs_other_religion = -0.1
				}
				start_value = -100
				end_value = 0
			}

			# Monthly drift. Stability, prestige, burghers, religious unity, clergy and war are in
			# common/static_modifiers/00_static_modifiers.txt and common/estates, so they show by name
			# Towards Paratge
			range_modifier = {
				trigger = {
					num_accepted_cultures = 2
				}
				modifier = {
					monthly_paratge = 0.1
				}
			}
			range_modifier = {
				trigger = {
					artist = 1
				}
				modifier = {
					monthly_paratge = 0.1
				}
			}
			range_modifier = {
				trigger = {
					faction_in_power = cathar_traders
				}
				modifier = {
					monthly_paratge = 0.1
				}
			}

			# Towards Zeal
			range_modifier = {
				trigger = {
					faction_in_power = cathar_church
				}
				modifier = {
					monthly_paratge = -0.1
				}
			}
			range_modifier = {
				trigger = {
					faction_in_power = cathar_militarists
				}
				modifier = {
					monthly_paratge = -0.05
				}
			}
			range_modifier = {
				trigger = {
					inquisitor = 1
				}
				modifier = {
					monthly_paratge = -0.1
				}
			}

			# At most once every 20 years each
			on_max_reached = {
				if = {
					limit = {
						OR = {
							NOT = { has_country_flag = occ_paratge_max_flag }
							had_country_flag = { flag = occ_paratge_max_flag days = 7300 }
						}
					}
					clr_country_flag = occ_paratge_max_flag
					set_country_flag = occ_paratge_max_flag
					country_event = { id = occ_events.7 }
				}
			}
			on_min_reached = {
				if = {
					limit = {
						OR = {
							NOT = { has_country_flag = occ_paratge_min_flag }
							had_country_flag = { flag = occ_paratge_min_flag days = 7300 }
						}
					}
					clr_country_flag = occ_paratge_min_flag
					set_country_flag = occ_paratge_min_flag
					country_event = { id = occ_events.8 }
				}
			}
		}
	}

	interactions = {
		# Paratge gain scales with prestige, doubled by an artist (like EE's Frankish Chivalry). Near the top it gives prestige instead
		occ_patronise_troubadours = {
			gui = occ_paratge_interaction
			icon = GFX_FRX_chansons_de_geste
			trigger = {
				dip_power_cost = 50
				years_of_income = 0.25
				prestige = 5
			}
			effect = {
				dip_power_cost = 50
				add_years_of_income = -0.25
				custom_tooltip = " "
				if = {
					limit = {
						has_government_power = {
							mechanic_type = occ_paratge_mechanic
							power_type = paratge
							value = 75
						}
					}
					custom_tooltip = occ_troubadours_overflow_tt
					add_prestige_or_monarch_power = { amount = 15 }
				}
				else = {
					custom_tooltip = occ_troubadours_scaling_tt
					hidden_effect = {
						occ_troubadours_paratge_effect = yes
					}
				}
			}
			cooldown_years = 10
			ai_chance = {
				factor = 100
			}
		}

		# Spends Paratge: refugee craftsmen add production to trade cities, minorities get a local modifier
		occ_open_the_gates = {
			gui = occ_paratge_interaction
			icon = GFX_occ_open_the_gates
			trigger = {
				has_government_power = {
					mechanic_type = occ_paratge_mechanic
					power_type = paratge
					value = 40
				}
			}
			effect = {
				if = {
					limit = {
						has_government_power = {
							mechanic_type = occ_paratge_mechanic
							power_type = paratge
							value = 80
						}
					}
					custom_tooltip = occ_open_gates_settle_3_tt
					hidden_effect = {
						occ_open_gates_settle_effect = yes
						occ_open_gates_settle_effect = yes
						occ_open_gates_settle_effect = yes
					}
					add_estate_loyalty = {
						estate = estate_burghers
						loyalty = 10
					}
				}
				else = {
					custom_tooltip = occ_open_gates_settle_2_tt
					hidden_effect = {
						occ_open_gates_settle_effect = yes
						occ_open_gates_settle_effect = yes
					}
				}
				hidden_effect = {
					every_owned_province = {
						limit = {
							has_province_flag = occ_gates_settled_flag
						}
						clr_province_flag = occ_gates_settled_flag
					}
				}
				every_owned_province = {
					limit = {
						is_city = yes
						OR = {
							has_owner_religion = no
							has_owner_accepted_culture = no
						}
					}
					add_province_modifier = {
						name = occ_open_gates_province_modifier
						duration = 3650
					}
				}
				add_government_power = {
					mechanic_type = occ_paratge_mechanic
					power_type = paratge
					value = -25
				}
			}
			cooldown_years = 10
			ai_chance = {
				factor = 50
			}
		}

		# Pulls the bar 25 back towards the middle. The reward depends on the side it is pulled from
		occ_public_disputation = {
			gui = occ_paratge_interaction
			icon = GFX_occ_public_disputation
			trigger = {
				OR = {
					has_government_power = {
						mechanic_type = occ_paratge_mechanic
						power_type = paratge
						value = 30
					}
					NOT = {
						has_government_power = {
							mechanic_type = occ_paratge_mechanic
							power_type = paratge
							value = -29
						}
					}
				}
			}
			effect = {
				if = {
					limit = {
						has_government_power = {
							mechanic_type = occ_paratge_mechanic
							power_type = paratge
							value = 0
						}
					}
					custom_tooltip = occ_disputation_paratge_tt
					add_country_modifier = {
						name = occ_disputation_tolerance_modifier
						duration = 3650
					}
					every_neighbor_country = {
						limit = {
							NOT = { religion = ROOT }
						}
						add_opinion = {
							who = ROOT
							modifier = occ_disputation_opinion
						}
					}
					add_government_power = {
						mechanic_type = occ_paratge_mechanic
						power_type = paratge
						value = -25
					}
				}
				else = {
					custom_tooltip = occ_disputation_zeal_tt
					if = {
						limit = {
							any_owned_province = {
								is_city = yes
								has_owner_religion = no
							}
						}
						custom_tooltip = occ_disputation_convert_tt
						hidden_effect = {
							occ_disputation_convert_effect = yes
						}
					}
					if = {
						limit = {
							has_estate = estate_church
						}
						add_estate_loyalty = {
							estate = estate_church
							loyalty = 10
						}
					}
					add_country_modifier = {
						name = occ_disputation_zeal_modifier
						duration = 3650
					}
					add_government_power = {
						mechanic_type = occ_paratge_mechanic
						power_type = paratge
						value = 25
					}
				}
			}
			cooldown_years = 10
			ai_chance = {
				factor = 25
			}
		}
	}
}
