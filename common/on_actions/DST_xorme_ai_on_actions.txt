# ROOT = New vassal, FROM = Overlord
# The OPM killer's conquest sphere stays on a country after it is vassalised and blocks diplomatic annexation
on_create_vassal = {
	if = {
		limit = {
			OR = {
				has_opinion_modifier = {
					who = FROM
					modifier = xorme_ai_conq_box
				}
				reverse_has_opinion_modifier = {
					who = FROM
					modifier = xorme_ai_conq_box
				}
			}
		}
		remove_opinion = {
			who = FROM
			modifier = xorme_ai_conq_box
		}
		reverse_remove_opinion = {
			who = FROM
			modifier = xorme_ai_conq_box
		}
	}
}
