# Kemetic clergy privileges (listed in common/estates/01_church.txt)
# Sacred Burials: offered by kemet_events.2 (faf_the_necropolis_of_bahiriya), always open to Kemet.
# The Houses of Life: unlocked by egp_the_house_of_life (missions/_DST_Kemet_Missions.txt).

kemet_sacred_burials_privilege = {
	icon = privilege_empty_throne
	max_absolutism = -10
	loyalty = 0.05
	influence = 0.15
	can_select = {
		religion = egyptian_religion
		OR = {
			government = monarchy
			government = tribal
		}
		OR = {
			mission_completed = faf_the_necropolis_of_bahiriya
			tag = EGP
		}
		any_owned_province = {
			region = egypt_region
			religion = egyptian_religion
		}
	}
	on_granted = {
		change_estate_land_share = {
			estate = estate_church
			share = 10
		}
		custom_tooltip = kemet_sacred_burials_privilege_tt
		hidden_effect = {
			if = {
				limit = {
					has_personal_deity = egy_anubis
					owns = 2318
				}
				add_country_modifier = {
					name = kemet_favour_of_anubis_modifier
					duration = -1
				}
			}
		}
	}
	benefits = {
		no_stability_loss_on_monarch_death = yes
	}
	is_valid = {
		religion = egyptian_religion
	}
	on_revoked = {
		add_stability = -2
		hidden_effect = {
			kemet_patron_favour_remove = { house = anubis }
		}
	}
	on_invalid = {
		kemet_patron_favour_remove = { house = anubis }
	}
	ai_will_do = {
		factor = 5
	}
}

kemet_house_of_life_privilege = {
	icon = privilege_aged_paper
	max_absolutism = -5
	loyalty = 0.05
	influence = 0.1
	is_valid = {
		religion = egyptian_religion
	}
	can_select = {
		religion = egyptian_religion
		mission_completed = egp_the_house_of_life
	}
	on_granted = {
		custom_tooltip = kemet_house_of_life_privilege_tt
	}
	benefits = {
		global_institution_spread = 0.1
		technology_cost = -0.03
	}
	ai_will_do = {
		factor = 10
	}
}
