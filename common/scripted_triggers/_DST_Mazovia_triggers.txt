# Mazovia on the vanilla map (1444.11.9 and later), where it starts as the Crown's vassal
# Missions: missions/_DST_Mazovia_Missions.txt, events: events/_DSE_flavour_Mazovia.txt
# The shattered starts use the Piast duchies tree instead (is_piast_duchy_trigger)

maz_vassal_tree_trigger = {
	tag = MAZ
	started_in = 1444.11.9
}

# The branch picked in maz_the_piotrkow_offer (maz_events.1), or previewed while the preview is open
maz_fealty_trigger = {
	OR = {
		has_country_flag = maz_crown_fealty
		AND = {
			has_country_flag = maz_preview_fealty
			can_preview_mission_of_key = { key = MZV }
		}
	}
}
maz_candidacy_trigger = {
	OR = {
		has_country_flag = maz_crown_candidacy
		AND = {
			has_country_flag = maz_preview_candidacy
			can_preview_mission_of_key = { key = MZV }
		}
	}
}

# The Crown: Poland, or the Commonwealth it becomes. A sworn Mazovia stays a subject when Poland forms PLC
# (decisions/TheCommonwealth.txt accepts it through maz_crown_settled_trigger), so the Fealty mechanics follow the new tag
maz_crown_subject_trigger = {
	custom_trigger_tooltip = {
		tooltip = maz_crown_subject_tt
		OR = {
			is_subject_of = POL
			is_subject_of = PLC
		}
	}
}

# Province scope: held by the Crown that Mazovia ($who$) serves
maz_held_by_crown_trigger = {
	owner = {
		OR = {
			tag = POL
			tag = PLC
		}
		$who$ = { is_subject_of = PREV }
	}
}

# Mazovia scope: a human Mazovia sworn to $crown$ that never broke the oath. Poland can't annex it (annexationaction in
# common/diplomatic_actions/00_diplomatic_actions.txt has the same conditions), so Poland's trees and the Commonwealth take it in place of an annexed Mazovia
maz_sworn_to_trigger = {
	ai = no
	has_country_flag = maz_crown_fealty
	NOT = { has_country_flag = maz_fealty_broken }
	is_subject_of = $crown$
}

# The Crown's side ($crown$: ROOT, or POL for Lithuania's Commonwealth decision): Mazovia is gone or is the Crown's sworn duchy.
# Replaces MAZ = { exists = no } in Poland's missions and decisions/TheCommonwealth.txt, so a sworn player Mazovia never blocks them
maz_crown_settled_trigger = {
	OR = {
		NOT = { exists = MAZ }
		custom_trigger_tooltip = {
			tooltip = maz_crown_settled_sworn_tt
			MAZ = { maz_sworn_to_trigger = { crown = $crown$ } }
		}
	}
}

# Country scope (Poland or the Commonwealth): our sworn Mazovia holds Warszawa, so we will never own it.
# Poland's Warszawa missions (missions/_ME_Poland_Missions.txt) then look at our capital instead
maz_crown_warszawa_sworn_trigger = {
	257 = { owned_by = MAZ }
	MAZ = { maz_sworn_to_trigger = { crown = ROOT } }
}

# Country scope (Poland or the Commonwealth): Warszawa is our core, or our sworn Mazovia holds it
maz_crown_holds_warszawa_trigger = {
	OR = {
		owns_core_province = 257
		custom_trigger_tooltip = {
			tooltip = maz_crown_warszawa_sworn_tt
			maz_crown_warszawa_sworn_trigger = yes
		}
	}
}

# Fiefs: a sworn Mazovia can be given almost any Crown province (maz_fief_eligible_trigger) and can't be annexed back.
# Where a Polish or Lithuanian mission or the Commonwealth needs a province built up or developed, these accept it in our sworn Mazovia's hands.
# The Mazovian branch only shows in the tooltip when it applies. Plain land checks use country_or_non_sovereign_subject_holds instead.

# Mazovia scope: a human Mazovia sworn to the Crown (Poland or the Commonwealth) that is $crown$ or $crown$'s overlord.
# The second case is Lithuania as the Crown's junior partner, whose land Mazovia may take as fiefs
maz_sworn_for_trigger = {
	ai = no
	has_country_flag = maz_crown_fealty
	NOT = { has_country_flag = maz_fealty_broken }
	overlord = {
		OR = {
			tag = POL
			tag = PLC
		}
		OR = {
			tag = $crown$
			$crown$ = { is_subject_of = PREV }
		}
	}
}

# Province scope: owned by $crown$, or by the sworn Mazovia of $crown$ or of its overlord (maz_sworn_for_trigger)
maz_owned_or_sworn_trigger = {
	if = {
		limit = {
			owned_by = MAZ
			MAZ = { maz_sworn_for_trigger = { crown = $crown$ } }
		}
		owned_by = MAZ
	}
	else = {
		owned_by = $crown$
	}
}

# Country scope ($crown$: ROOT, or POL for Lithuania's Commonwealth decision): owns_core_province = $province$, or a sworn Mazovia as above holds it
maz_owns_core_or_sworn_trigger = {
	if = {
		limit = {
			$province$ = { owned_by = MAZ }
			MAZ = { maz_sworn_for_trigger = { crown = $crown$ } }
		}
		$province$ = { owned_by = MAZ }
	}
	else = {
		owns_core_province = $province$
	}
}

# A loyal vassal of the Crown: Poland's subject with little wish to leave, that never declared war on it
# (maz_fealty_broken: declarewar effect in common/diplomatic_actions/00_diplomatic_actions.txt)
maz_loyal_vassal_trigger = {
	maz_crown_subject_trigger = yes
	NOT = { liberty_desire = 25 }
	custom_trigger_tooltip = {
		tooltip = maz_kept_the_oath_tt
		NOT = { has_country_flag = maz_fealty_broken }
	}
}

# A sworn Mazovia that can pad its blood for the Crown with forged muster rolls (maz_forge_rolls_effect).
# Gates the Mazovia options in events/ArmyProfessionalism.txt (.4 .5 .9 .11) and the clerk, maz_events.10
maz_can_forge_rolls_trigger = {
	has_country_flag = maz_crown_fealty
	NOT = { has_country_flag = maz_fealty_broken }
	maz_crown_subject_trigger = yes
}

# The 5-year pulse events about blood for the Crown (maz_events.11 to .14, common/on_actions/DST_Mazovia_on_actions.txt):
# each fires at most once every 30 years. $flag$ is set again in the event's immediate
maz_pulse_event_ready_trigger = {
	maz_can_forge_rolls_trigger = yes
	OR = {
		NOT = { has_country_flag = $flag$ }
		had_country_flag = {
			flag = $flag$
			days = 10950
		}
	}
}

# Form Poland (decisions/PolishNation.txt) while Poland still exists: only after maz_crown_to_warszawa,
# with Poland as our subject. The decision inherits Poland before changing tag.
maz_inherits_poland_trigger = {
	has_country_flag = maz_crown_to_warszawa_flag
	exists = POL
	POL = { is_subject_of = ROOT }
}

# Province scope: in the same map area as its owner's capital. Not is_in_capital_area, which is the capital's whole
# landmass (the AI uses it for overseas wars): it made nearly every mainland province of the Crown unclaimable
# (Sandomierz, user 2026-11-16). Pattern from vanilla buildings (courthouse, area_for_scope_province)
maz_in_owner_capital_area_trigger = {
	area_for_scope_province = {
		is_capital = yes
		owned_by = PREV
	}
}

# Province scope: a province Mazovia ($who$) may claim as a fief of the Crown. Held by Poland or another of Poland's subjects,
# not in the area of its owner's capital, not Krakow or Gniezno (the coronation towns)
maz_fief_eligible_trigger = {
	is_city = yes
	NOT = { owned_by = $who$ }
	owner = {
		OR = {
			tag = POL
			is_subject_of = POL
			tag = PLC
			is_subject_of = PLC
		}
	}
	NOT = { maz_in_owner_capital_area_trigger = yes }
	NOT = { province_id = 262 }
	NOT = { province_id = 4526 }
}

#### The coronation: Mazovia's shop in the shared tier menu (common/scripted_effects/DST_tier_menu_effects.txt),
#### shop maz_coronation. Effects: common/scripted_effects/DST_Mazovia_scripted_effects.txt

# Country scope: all 4 x 3 lines exist
maz_coronation_line_exists_trigger = {
	OR = { # always true. Both arguments must appear in the body: the engine refuses a call whose arguments are never used
		is_variable_equal = {
			which = dst_tier_menu_group$group$_line
			value = $line$
		}
		always = yes
	}
}

# Country scope: our blood covers the price of everything chosen (dst_tier_menu_cost)
maz_coronation_can_afford_trigger = {
	custom_trigger_tooltip = {
		tooltip = maz_coronation_afford_tt
		check_variable = {
			which = maz_crown_blood
			which = dst_tier_menu_cost
		}
	}
}

# Country scope: too little blood for even one privilege at its first degree (2500, maz_coronation_price_t1_effect),
# so the coronation may be confirmed with nothing chosen
maz_coronation_nothing_affordable_trigger = {
	NOT = {
		check_variable = {
			which = maz_crown_blood
			value = 2500
		}
	}
}

# Country scope: the Mazovian Election (mission maz_the_mazovian_election) can be held. Oath kept and still the Crown's subject
# (not maz_loyal_vassal_trigger: a Mazovia half the Crown's size has high liberty desire by design), an AI Crown (an AI Mazovia
# never takes over a player's Poland), the Crown at peace, its opinion of us 150, its succession in doubt, and our development
# at least half of its own (maz_half_the_crown, set monthly by maz_half_the_crown_effect)
maz_election_ready_trigger = {
	custom_trigger_tooltip = {
		tooltip = maz_election_sworn_tt
		maz_can_forge_rolls_trigger = yes
	}
	if = { # no overlord scope to read once we are free
		limit = {
			is_subject = yes
		}
		custom_trigger_tooltip = {
			tooltip = maz_election_crown_ai_tt
			overlord = { ai = yes }
		}
		custom_trigger_tooltip = {
			tooltip = maz_election_crown_peace_tt
			overlord = { is_at_war = no }
		}
		custom_trigger_tooltip = {
			tooltip = maz_election_opinion_tt
			overlord = {
				has_opinion = {
					who = ROOT
					value = 150
				}
			}
		}
		custom_trigger_tooltip = {
			tooltip = maz_election_crisis_tt
			overlord = {
				OR = {
					has_regency = yes
					has_heir = no
				}
			}
		}
	}
	custom_trigger_tooltip = {
		tooltip = maz_election_half_tt
		has_country_flag = maz_half_the_crown
	}
}
