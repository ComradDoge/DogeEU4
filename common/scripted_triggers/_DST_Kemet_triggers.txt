# Farafra (FAF) and Kemet (EGP): shared triggers
# Missions: missions/_DST_Farafra_Missions.txt, missions/_DST_Kemet_Missions.txt
# Effects: common/scripted_effects/DST_Kemet_scripted_effects.txt

# At least $amount$ of the eight Houses of the Gods restored (decisions/_DST_flavour_Kemet_Houses.txt).
# Wrap in custom_trigger_tooltip: the raw form lists all eight flags.
kemet_houses_restored_trigger = {
	calc_true_if = {
		amount = $amount$
		has_country_flag = kemet_house_sutekh
		has_country_flag = kemet_house_heru
		has_country_flag = kemet_house_aset
		has_country_flag = kemet_house_wsir
		has_country_flag = kemet_house_amun
		has_country_flag = kemet_house_ptah
		has_country_flag = kemet_house_sobek
		has_country_flag = kemet_house_djehuty
	}
}

# The land the Kemet formation asks for (decisions/_DST_FormAncientEgypt.txt).
kemet_holds_formation_lands_trigger = {
	num_of_owned_provinces_with = {
		value = 20
		region = egypt_region
		is_core = ROOT
	}
}

# A country that has chosen its Faith of the River (kemet_events.1).
kemet_faith_chosen_trigger = {
	OR = {
		has_country_flag = kemet_path_maat
		has_country_flag = kemet_path_neteru
	}
}

# The path the Farafra missions follow: the chosen one, or the one previewed in the mission view (key FAF,
# EE_scripted_effects_mission_rewards.txt) before it is confirmed. Events and effects read the real flags.
kemet_path_maat_trigger = {
	OR = {
		has_country_flag = kemet_path_maat
		AND = {
			has_country_flag = kemet_preview_maat
			can_preview_mission_of_key = { key = FAF }
		}
	}
}
kemet_path_neteru_trigger = {
	OR = {
		has_country_flag = kemet_path_neteru
		AND = {
			has_country_flag = kemet_preview_neteru
			can_preview_mission_of_key = { key = FAF }
		}
	}
}
