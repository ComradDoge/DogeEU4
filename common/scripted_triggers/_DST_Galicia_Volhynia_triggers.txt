# Galicia and Volhynia (GLA, VLH): the Crown of Rus'
# Missions: missions/_MIS_Galicia_Volhynia_Missions.txt, events: events/_DSE_flavour_Galicia_Volhynia.txt
# Effects: common/scripted_effects/DST_Galicia_Volhynia_scripted_effects.txt

# The two Ruthenian successor states that share the Crown of Rus' tree
gv_is_galicia_volhynia_trigger = {
	OR = {
		tag = GLA
		tag = VLH
	}
}

# The crown the missions follow: the one chosen in gv_events.3, or the one previewed in the mission view (key GV,
# EE_scripted_effects_mission_rewards.txt) before it is confirmed. Events and effects read the real flags.
gv_crown_of_rome_trigger = {
	OR = {
		has_country_flag = gv_crown_of_rome
		AND = {
			has_country_flag = gv_preview_rome
			can_preview_mission_of_key = { key = GV }
		}
	}
}
gv_crown_of_halicz_trigger = {
	OR = {
		has_country_flag = gv_crown_of_halicz
		AND = {
			has_country_flag = gv_preview_halicz
			can_preview_mission_of_key = { key = GV }
		}
	}
}

# The column tails below the crown missions, shared by both crown series (gv_c2_rome / gv_c2_halicz, gv_c3_*).
# Each tail exists once per crown so its required_missions arrow can point at that crown's mission.
gv_the_chronicle_trigger = {
	stability = 2
	OR = {
		adm = 3
		has_adm_advisor_2 = yes
	}
}
gv_ruska_pravda_trigger = {
	reform_level = 3
	total_development = 150
}
gv_rex_russiae_trigger = {
	custom_trigger_tooltip = {
		tooltip = gv_crowned_ten_years_tt
		had_country_flag = {
			flag = gv_crowned_flag
			days = 3650
		}
	}
	total_development = 250
	OR = {
		custom_trigger_tooltip = {
			tooltip = gv_rex_russiae_princes_tt
			calc_true_if = {
				all_subject_country = {
					culture_group = east_slavic
					NOT = { is_subject_of_type = tributary_state }
				}
				amount = 2
			}
		}
		total_development = 350
	}
}
gv_lublin_and_sandomierz_trigger = {
	sandomierz_area = {
		type = all
		country_or_non_sovereign_subject_holds = ROOT
	}
}
