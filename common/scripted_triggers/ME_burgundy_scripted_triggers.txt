# Burgundy/Lotharingia (or what they became) picked the Europa Expanded mission tree
# Only offered through the Emperor DLC mission tree choice; without Emperor the vanilla Burgundian missions stay.
ME_bur_uses_ee_missions_trigger = {
	OR = {
		tag = BUR
		tag = LOT
		was_tag = BUR
		was_tag = LOT
	}
	has_country_flag = use_missions_expanded_missions_instead_of_paradox
}

# Province scope (Brussels): Burgundian discount on the Coudenberg Palace is active (until it reaches tier 1)
ME_bur_coudenberg_discount_active_trigger = {
	has_province_modifier = ME_bur_coudenberg_patronage
}
