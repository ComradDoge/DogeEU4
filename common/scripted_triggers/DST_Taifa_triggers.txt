# The Taifas (SEV, ALC, ALN): the road to al-Andalus
# Missions: missions/_MIS_Taifa_Missions.txt, events: events/_DSE_flavour_Taifas.txt
# Effects: common/scripted_effects/DST_Taifa_scripted_effects.txt

# The three Andalusi taifas that share the taifa tree (Granada has its own ME tree)
taifa_is_taifa_trigger = {
	OR = {
		tag = SEV
		tag = ALC
		tag = ALN
	}
}

# Every Andalusi taifa, Granada included: for the rivals the taifa tree targets, not for who gets the tree
taifa_is_andalusi_taifa_trigger = {
	OR = {
		tag = SEV
		tag = ALC
		tag = ALN
		tag = GRA
	}
}

# Owns every province the Andalusia decision asks for (decisions/AndalusianNation.txt)
taifa_owns_al_andalus_trigger = {
	alentejo_area = {
		type = all
		owned_by = ROOT
	}
	lower_andalucia_area = {
		type = all
		owned_by = ROOT
	}
	upper_andalucia_area = {
		type = all
		owned_by = ROOT
	}
	toledo_area = {
		type = all
		owned_by = ROOT
	}
	extremadura_area = {
		type = all
		owned_by = ROOT
	}
}

# Province scope: in one of the five areas the Andalusia decision asks for (the Gardens of al-Andalus column)
taifa_in_al_andalus_trigger = {
	OR = {
		area = alentejo_area
		area = lower_andalucia_area
		area = upper_andalucia_area
		area = toledo_area
		area = extremadura_area
	}
}

# Province scope: a field the New Crops can be planted in (decision taifa_plant_the_new_crops, taifa_events.20)
taifa_new_crops_field_trigger = {
	region = iberia_region
	OR = {
		trade_goods = grain
		trade_goods = livestock
		trade_goods = wool
		trade_goods = wine
	}
	NOT = { has_manufactory_trigger = yes }
	NOT = { has_latent_trade_goods = coal }	# the coal fields of the Industrial age
	NOT = { has_province_flag = has_latent_good }	# TGE factory province
}
