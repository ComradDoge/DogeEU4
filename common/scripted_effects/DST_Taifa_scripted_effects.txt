# The Taifas (SEV, ALC, ALN): the road to al-Andalus
# Missions: missions/_MIS_Taifa_Missions.txt, events: events/_DSE_flavour_Taifas.txt
# Triggers: common/scripted_triggers/DST_Taifa_triggers.txt

# Saves the Maghrebi court the taifa can call on as event_target:taifa_maghrebi_court (taifa_events.2):
# Tetouan, Fez or Marrakesh if one stands free, else any free Muslim country ruling from the Maghreb.
# Sets taifa_maghrebi_court_found when one exists, so the event can switch its text.
taifa_save_maghrebi_court_effect = {
	clr_country_flag = taifa_maghrebi_court_found
	if = {
		limit = {
			TET = {
				exists = yes
				is_subject = no
				religion_group = muslim
				NOT = { war_with = ROOT }
			}
		}
		TET = { save_event_target_as = taifa_maghrebi_court }
		set_country_flag = taifa_maghrebi_court_found
	}
	else_if = {
		limit = {
			FEZ = {
				exists = yes
				is_subject = no
				religion_group = muslim
				NOT = { war_with = ROOT }
			}
		}
		FEZ = { save_event_target_as = taifa_maghrebi_court }
		set_country_flag = taifa_maghrebi_court_found
	}
	else_if = {
		limit = {
			MRK = {
				exists = yes
				is_subject = no
				religion_group = muslim
				NOT = { war_with = ROOT }
			}
		}
		MRK = { save_event_target_as = taifa_maghrebi_court }
		set_country_flag = taifa_maghrebi_court_found
	}
	else_if = {
		limit = {
			any_country = {
				is_subject = no
				religion_group = muslim
				NOT = { tag = ROOT }
				NOT = { war_with = ROOT }
				capital_scope = {
					region = maghreb_region
				}
			}
		}
		random_country = {
			limit = {
				is_subject = no
				religion_group = muslim
				NOT = { tag = ROOT }
				NOT = { war_with = ROOT }
				capital_scope = {
					region = maghreb_region
				}
			}
			save_event_target_as = taifa_maghrebi_court
		}
		set_country_flag = taifa_maghrebi_court_found
	}
}
