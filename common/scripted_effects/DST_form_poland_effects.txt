# What forming Poland does once the country carries the POL tag: used by the Form Poland decision
# (decisions/PolishNation.txt, after its change_tag) and by the Mazovian Election (maz_election_effect,
# common/scripted_effects/DST_Mazovia_scripted_effects.txt). No square brackets in this file.
dst_form_poland_effect = {
	on_change_tag_effect = yes
	if = { # AI never gets the mission tree choice after startup, default it to the Paradox tree
		limit = {
			ai = yes
			NOT = { has_country_flag = use_paradox_missions }
			NOT = { has_country_flag = use_missions_expanded_missions_instead_of_paradox }
		}
		set_country_flag = use_paradox_missions
	}
	swap_non_generic_missions = yes
	if = {
		limit = { has_dlc = "Lions of the North" }
		complete_mission = pol_leverage_magnates
		complete_mission = pol_great_sejm
		complete_mission = plc_articles_agreement
		complete_mission = pol_varna_aftermath
	}
	add_prestige_or_monarch_power = { amount = 15 }


	# Flags Hidden
	set_country_flag = formed_poland_flag


	# Other
	#custom_tooltip = tooltip_line_seperate_other


	# Government
	custom_tooltip = tooltip_line_seperate_government
	if = {
		limit = {
			NOT = { government_rank = 2 }
		}
		set_government_rank = 2
	}


	#### Gov Reform
	#custom_tooltip = tooltip_line_seperate_gov_gap


	# Culture
	custom_tooltip = tooltip_line_seperate_culture
	if = {
		limit = {
			NOT = {
				primary_culture = polish
			}
		}
		change_primary_culture = polish
	}

	# Claims
	custom_tooltip = tooltip_line_seperate_claims
	poland_region = {
		limit = {
			NOT = { owned_by = ROOT }
		}
		add_permanent_claim = POL
	}

	# Events
	custom_tooltip = tooltip_line_seperate_events
	if = {
		limit = { has_custom_ideas = no }
		country_event = { id = ideagroups.1 } #Swap Ideas
	}


	# Modifiers
	custom_tooltip = tooltip_line_seperate_modifiers
	add_country_modifier = {
		name = "centralization_modifier"
		duration = 7300
	}

	# Hidden Effects
	hidden_effect = {
		if = { # Piast duchies: the Seniorate ends with the crown (common/scripted_effects/DST_Piast_scripted_effects.txt)
			limit = {
				has_country_flag = piast_senior_duke
			}
			piast_vacate_seniorate_effect = yes
		}
	}
}
