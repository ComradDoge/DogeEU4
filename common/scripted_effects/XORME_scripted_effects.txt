#DEVELOPMENT ( Used in Buildings for AI forts )
xorme_ai_add_dev_1_effect = {
	add_base_tax = 1
	add_base_production = 1
	add_base_manpower = 1
}

xorme_ai_remove_dev_1_effect = {
	add_base_tax = -1
	add_base_production = -1
	add_base_manpower = -1
}

#ESTATES
xorme_ai_estates_initialization_effect = {
	if = {
		limit = { has_estate = estate_burghers capital_scope = { continent = africa } gold = 1 }
		set_estate_privilege = estate_burghers_control_over_the_mint
	}
	if = {
		limit = { has_estate = xrome_ai_estate }
		set_estate_privilege = estate_xorme_ai_supremacy_over_crown
	}
}

xorme_ai_create_court = {
	if = {
		limit = { num_of_ports = 1 }
		create_admiral = {
			tradition = 25
		}
	}
	create_general = {
		tradition = 10
	}
	add_mil_power = -35
	add_dip_power = -35
}

xorme_ai_set_crown_land = {
	if = {
		limit = { NOT = { crown_land_share = 25 } }
		change_estate_land_share = {
			estate = all
			share = -30
		}
	}
}

#BUILD FORT
xorme_ai_build_forts_effect = {
	if = {
		limit = {
			can_build = fort_18th
			FROM = {
				treasury = 1600
				monthly_income = 50
				current_income_balance = 25
			}
		}
		add_building_construction = {
			building = fort_18th
			speed = 1
			cost = 0.75
		}
	}
	else_if = {
		limit = {
			can_build = fort_17th
			FROM = {
				treasury = 800
				monthly_income = 30
				current_income_balance = 15
			}
		}
		add_building_construction = {
			building = fort_17th
			speed = 1
			cost = 0.75
		}
	}
	else_if = {
		limit = {
			can_build = fort_16th
			FROM = {
				treasury = 400
				monthly_income = 20
				current_income_balance = 10
			}
		}
		add_building_construction = {
			building = fort_16th
			speed = 1
			cost = 0.75
		}
	}
	else_if = {
		limit = {
			can_build = fort_15th
			FROM = {
				treasury = 200
				monthly_income = 12
				current_income_balance = 6
			}
		}
		add_building_construction = {
			building = fort_15th
			speed = 1
			cost = 0.75
		}
	}
	else = {}
}

#OPM KILLER ( Used in xorme_ai_opm_killer.1 )
#every_neighbor_country never reaches island realms, so small AI islands close to ROOT join its conquest sphere here
xorme_ai_conq_box_island_minors_effect = {
	every_country = {
		limit = {
			exists = yes
			ai = yes
			NOT = { tag = ROOT }
			NOT = { total_development = $dev$ }
			NOT = { is_neighbor_of = ROOT }
			NOT = { is_subject_of = ROOT }
			[[exclude_allies]
				NOT = { alliance_with = ROOT }
			]
			NOT = { historical_friend_with = ROOT }
			NOT = { tag = PAP }
			OR = {
				capital_scope = { is_island = yes }
				NOT = { any_neighbor_country = { exists = yes } }
			}
			NOT = {
				border_distance = {
					who = ROOT
					distance = 150
				}
			}
		}
		add_opinion = {
			who = ROOT
			modifier = xorme_ai_conq_box
		}
		reverse_add_opinion = {
			who = ROOT
			modifier = xorme_ai_conq_box
		}
	}
}
