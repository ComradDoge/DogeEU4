# Livonia refounds the Livonian Order (dse_flavour_livonia.1)
# Set the liv_*_path flag before calling, so the mission swap picks the right branch.
lva_refound_livonian_order_effect = {
	custom_tooltip = tooltip_line_seperate_general
	change_tag = LIV
	on_change_tag_effect = yes

	custom_tooltip = tooltip_line_seperate_government
	change_government_to_theocracy = yes
	add_government_reform = monastic_order_reform

	custom_tooltip = tooltip_line_seperate_culture
	if = {
		limit = {
			NOT = { primary_culture = latvian }
			NOT = { accepted_culture = latvian }
		}
		add_accepted_culture = latvian
	}
	if = {
		limit = {
			NOT = { primary_culture = estonian }
			NOT = { accepted_culture = estonian }
		}
		add_accepted_culture = estonian
	}

	custom_tooltip = tooltip_line_seperate_events
	swap_non_generic_missions = yes
	country_event = { id = dse_flavour_name_change.1 }
	if = {
		limit = { has_custom_ideas = no }
		country_event = { id = ideagroups.1 } #Swap Ideas
	}
}
