#### Helpers

# Kingdom (or khanate) rank. cum_crowned_flag marks the crowning (cum_the_crown_of_the_cumans).
cum_crown_effect = {
	if = {
		limit = {
			NOT = { government_rank = 2 }
		}
		set_government_rank = 2
	}
	hidden_effect = {
		set_country_flag = cum_crowned_flag
	}
}

# Back to horse (dse_flavour_cumania.1): every infantry regiment is swapped for a cavalry regiment in the same province, for free.
# Capped at 50 per province.
cum_infantry_to_cavalry_effect = {
	custom_tooltip = cum_infantry_to_cavalry_tt
	hidden_effect = {
		every_owned_province = {
			limit = {
				infantry_in_province = ROOT
			}
			set_variable = {
				which = cum_inf_swap
				value = 0
			}
			while = {
				limit = {
					infantry_in_province = ROOT
					NOT = {
						check_variable = {
							which = cum_inf_swap
							value = 50
						}
					}
				}
				kill_units = {
					who = ROOT
					type = infantry
					amount = 1
				}
				cavalry = ROOT
				change_variable = {
					which = cum_inf_swap
					value = 1
				}
			}
			set_variable = {
				which = cum_inf_swap
				value = 0
			}
		}
	}
}

# Estate loyalty, skipped when the country lacks the estate (hordes have no nobles or burghers)
cum_estate_loyalty_effect = {
	if = {
		limit = {
			has_estate = $estate$
		}
		add_estate_loyalty = {
			estate = $estate$
			loyalty = $loyalty$
		}
	}
}

# The clans' favour, read once by dse_flavour_cumania.14 (The Clans Answer); no effect or tooltip after it
cum_clan_favour_up_effect = {
	if = {
		limit = {
			NOT = { mission_completed = cum_the_clans_answer }
		}
		custom_tooltip = cum_clan_favour_up_tt
		hidden_effect = {
			change_variable = {
				which = cum_clan_favour
				value = 1
			}
		}
	}
}

cum_clan_favour_down_effect = {
	if = {
		limit = {
			NOT = { mission_completed = cum_the_clans_answer }
		}
		custom_tooltip = cum_clan_favour_down_tt
		hidden_effect = {
			subtract_variable = {
				which = cum_clan_favour
				value = 1
			}
		}
	}
}
