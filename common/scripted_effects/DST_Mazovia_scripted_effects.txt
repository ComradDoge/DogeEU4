# Province scope: Mazovian settlers in the southern Prussian woods (maz_the_orders_border)
maz_masurian_settlers_effect = {
	if = {
		limit = {
			NOT = { culture = polish }
		}
		change_culture = polish
	}
	add_province_modifier = {
		name = maz_masurian_settlers_modifier
		duration = -1

#### Fiefs of the Crown: blood spilled for the Crown buys fiefs (maz_fiefs_of_the_crown unlocks them with maz_fiefs_unlocked).
#### Humans claim with the province buttons in common/custom_gui/DST_Mazovia_fiefs_gui.txt; the AI claims in maz_fief_monthly_effect.
#### Claiming a fief spends its price from maz_crown_blood. No wait between fiefs: the price is the only limit (user, 2026-11-12).

# Blood counts only Mazovia's own losses. The battle on_actions give the whole side's losses (land_units_killed is ROOT's side,
# names swapped by the engine), so a battle is settled in three steps:
# 1. maz_add_crown_blood_effect (country on_actions) adds the side's losses to maz_pending_blood and schedules maz_events.98 for the next day
# 2. maz_crown_blood_battle_share_effect (on_battle_won_province) counts our regiments and all allied regiments in the battle province
# 3. maz_events.98 adds pending x our regiments / allied regiments (us included) to maz_crown_blood, only if we had at least 4 regiments there
# The next-day settlement makes the order of the two on_actions irrelevant.
# No loc commands (square brackets) in scripted effects, not even in log strings: brackets are the parameter syntax and crash the load
# with a stack overflow. The logs with values are in maz_events.98

# on_battle_won_country / on_battle_lost_country (common/on_actions/DST_Mazovia_on_actions.txt). ROOT = a country in the battle, FROM = its enemy
maz_add_crown_blood_effect = {
	if = {
		limit = {
			has_country_flag = maz_crown_fealty
			NOT = { has_country_flag = maz_fealty_broken }
			maz_crown_subject_trigger = yes
			overlord = { war_with = FROM }
		}
		change_variable = {
			which = maz_pending_blood
			which = land_units_killed
		}
		if = {
			limit = {
				NOT = { has_country_flag = maz_blood_settling }
			}
			set_country_flag = maz_blood_settling
			country_event = {
				id = maz_events.98
				days = 1
			}
		}
	}
}

# on_battle_won_province, ROOT = the battle province: our regiments and every allied regiment here, added to MAZ's
# maz_battle_own / maz_battle_side. An ally has regiments here, is not at war with us and is at war with one of our enemies
# (us included), so neutral armies passing through on military access are left out. Runs for every battle in the world,
# so the limit comes first
maz_crown_blood_battle_share_effect = {
	if = {
		limit = {
			MAZ = {
				has_country_flag = maz_crown_fealty
				NOT = { has_country_flag = maz_fealty_broken }
				maz_crown_subject_trigger = yes
			}
			num_of_units_in_province = {
				who = MAZ
				amount = 1
			}
		}
		save_event_target_as = maz_battle_location
		set_variable = {
			which = maz_battle_own
			value = 0
		}
		set_variable = {
			which = maz_battle_side
			value = 0
		}
		maz_count_regiments_effect = {
			who = MAZ
			var = maz_battle_own
		}
		every_country = {
			limit = {
				event_target:maz_battle_location = {
					num_of_units_in_province = {
						who = PREV
						amount = 1
					}
				}
				NOT = { war_with = MAZ }
				any_country = {
					war_with = MAZ
					war_with = PREV
				}
			}
			event_target:maz_battle_location = {
				maz_count_regiments_effect = {
					who = PREV
					var = maz_battle_side
				}
			}
		}
		MAZ = {
			change_variable = {
				which = maz_battle_own
				which = PREV
			}
			change_variable = {
				which = maz_battle_side
				which = PREV
			}
		}
	}
}

# on_monthly_pulse, any country: keep Mazovia's fief prices current (development, borders and ownership change), and let an AI
# Mazovia claim a fief its blood pays for, at most one every 2 years (maz_fief_ai_cooldown, set in maz_fief_ai_claim_effect)
maz_fief_monthly_effect = {
	# The Mazovian Election (maz_election_ready_trigger) asks for half the Crown's development. Checked for every sworn
	# Mazovia, fiefs unlocked or not, so the election tooltip is right from the oath on
	if = {
		limit = {
			maz_can_forge_rolls_trigger = yes
		}
		maz_half_the_crown_effect = yes
	}
	if = {
		limit = {
			has_country_flag = maz_fiefs_unlocked
			NOT = { has_country_flag = maz_elected_king } # the elected king is the Crown: nothing left to price or claim
		}
		maz_fief_prices_refresh_effect = { who = ROOT }
		if = {
			limit = {
				had_country_flag = {
					flag = maz_fief_ai_cooldown
					days = 730
				}
			}
			clr_country_flag = maz_fief_ai_cooldown
		}
		if = {
			limit = {
				ai = yes
				maz_crown_subject_trigger = yes
				is_at_war = no
				NOT = { has_country_flag = maz_fealty_broken }
				NOT = { has_country_flag = maz_fief_ai_cooldown }
			}
			maz_fief_ai_claim_effect = yes
		}
	}
}

# Province scope, a fief candidate ($who$ = Mazovia): price it and mark whether our blood covers it.
# Cost = development x 500, x1.5 mountains, x1.25 hills/highlands/forest/woods/marsh/jungle, x2 if it does not border us,
# x0.75 while Poland's opinion of us is at least 150, x0.25 if it is our core, else x0.75 if we claim it.
# Our blood total is copied onto the province for the comparison and the tooltip
maz_fief_cost_effect = {
	export_to_variable = {
		which = maz_fief_cost
		value = development
	}
	multiply_variable = {
		which = maz_fief_cost
		value = 500
	}
	if = {
		limit = {
			has_terrain = mountain
		}
		multiply_variable = {
			which = maz_fief_cost
			value = 1.5
		}
	}
	else_if = {
		limit = {
			OR = {
				has_terrain = hills
				has_terrain = highlands
				has_terrain = forest
				has_terrain = woods
				has_terrain = marsh
				has_terrain = jungle
			}
		}
		multiply_variable = {
			which = maz_fief_cost
			value = 1.25
		}
	}
	if = {
		limit = {
			NOT = {
				any_neighbor_province = {
					owned_by = $who$
				}
			}
		}
		multiply_variable = {
			which = maz_fief_cost
			value = 2
		}
	}
	if = {
		limit = {
			OR = {
				POL = {
					has_opinion = {
						who = $who$
						value = 150
					}
				}
				PLC = {
					has_opinion = {
						who = $who$
						value = 150
					}
				}
			}
		}
		multiply_variable = {
			which = maz_fief_cost
			value = 0.75
		}
	}
	if = {
		limit = {
			is_core = $who$
		}
		multiply_variable = {
			which = maz_fief_cost
			value = 0.25
		}
	}
	else_if = {
		limit = {
			is_claim = $who$
		}
		multiply_variable = {
			which = maz_fief_cost
			value = 0.75
		}
	}
	set_variable = {
		which = maz_crown_blood
		which = $who$
	}
	set_province_flag = maz_fief_priced
	if = {
		limit = {
			check_variable = {
				which = maz_crown_blood
				which = maz_fief_cost
			}
		}
		set_province_flag = maz_fief_affordable
	}
	else = {
		clr_province_flag = maz_fief_affordable
	}
}

# Any scope: price every fief candidate of the Crown (Poland, or the Commonwealth it becomes) and its other subjects for
# Mazovia ($who$). Mazovia is passed, not taken from ROOT: from a province button ROOT is the province, and the old
# ROOT-based refresh then skipped every subject of the Crown and showed them a stale blood total (Hungary, 2026-11-16)
maz_fief_prices_refresh_effect = {
	every_country = {
		limit = {
			OR = {
				tag = POL
				tag = PLC
			}
		}
		every_owned_province = {
			limit = {
				maz_fief_eligible_trigger = { who = $who$ }
			}
			maz_fief_cost_effect = { who = $who$ }
		}
		every_subject_country = {
			limit = {
				NOT = { tag = $who$ }
			}
			every_owned_province = {
				limit = {
					maz_fief_eligible_trigger = { who = $who$ }
				}
