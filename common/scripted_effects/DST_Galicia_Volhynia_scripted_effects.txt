# Permanent claims on whatever ROOT doesn't own of the old Romanovich heartland:
# Red Ruthenia, Volhynia and Podolia (281). Used by gv_the_romanovich_inheritance.
gv_claim_the_other_half_effect = {
	red_ruthenia_area = {
		limit = {
			NOT = { owned_by = ROOT }
		}
		add_permanent_claim = ROOT
	}
	volhynia_area = {
		limit = {
			NOT = { owned_by = ROOT }
		}
		add_permanent_claim = ROOT
	}
	281 = {
		if = {
			limit = {
				NOT = { owned_by = ROOT }
			}
			add_permanent_claim = ROOT
		}
	}
}

# Kingdom rank and the Galicia-Volhynia name. Folded from the old galicia_volhynia_nameChange decision
# (decisions/_DST_Flavour.txt), so it keeps that decision's flag and name keys. on_change_tag_effect restores
# the name when the country later forms Ruthenia or Kievan Rus. gv_crowned_flag dates the coronation (gv_rex_russiae).
gv_coronation_effect = {
	if = {
		limit = {
			NOT = { government_rank = 2 }
		}
		set_government_rank = 2
	}
	if = {
		limit = {
			NOT = { has_country_flag = galicia_volhynia_namechange_flag }
		}
		custom_tooltip = gv_coronation_name_tt
		hidden_effect = {
			override_country_name = GALICIA_VOLHYNIA
			set_country_flag = galicia_volhynia_namechange_flag
		}
	}
	hidden_effect = {
		set_country_flag = gv_crowned_flag
	}
}
