# Changes Paratge (common/government_mechanics/DST_occ_paratge.txt). Negative values move it towards Zeal.
# occ_change_paratge = { value = 10 }
# Does nothing without the mechanic (a revolution sweeps Paratge away, occ_events.19)
occ_change_paratge = {
	if = {
		limit = {
			has_government_mechanic = occ_paratge_mechanic
		}
		add_government_power = {
			mechanic_type = occ_paratge_mechanic
			power_type = paratge
			value = $value$
		}
	}
}

# Country scope. Avignon Question, spare the palace (occ_events.4.b and its mission preview): a Palace of the Popes tier, +1 base tax once it is complete
occ_avignon_palace_effect = {
	202 = {
		if = {
			limit = {
				owned_by = ROOT
				NOT = {
					has_great_project = {
						type = palace_of_the_popes
						tier = 3
					}
				}
			}
			add_great_project_tier = {
				type = palace_of_the_popes
				tier = 1
			}
		}
		else_if = {
			limit = { owned_by = ROOT }
			add_base_tax = 1
		}
	}
}

# Country scope, immediate of occ_events.9. Picks a Cathar province for a House of the Good Men, 15+ development first
# occ_pick_good_men_house = { n = 1 } saves event target occ_good_men_house_1 (flag occ_good_men_house_pick, cleared by the event)
occ_pick_good_men_house = {
	if = {
		limit = {
			any_owned_province = {
				religion = cathar
				development = 15
				NOT = { has_province_modifier = occ_good_men_house_modifier }
				NOT = { has_province_flag = occ_good_men_house_pick }
			}
		}
		random_owned_province = {
			limit = {
				religion = cathar
				development = 15
				NOT = { has_province_modifier = occ_good_men_house_modifier }
				NOT = { has_province_flag = occ_good_men_house_pick }
			}
			set_province_flag = occ_good_men_house_pick
			save_event_target_as = occ_good_men_house_$n$
		}
	}
	else = {
		random_owned_province = {
			limit = {
				religion = cathar
				NOT = { has_province_modifier = occ_good_men_house_modifier }
				NOT = { has_province_flag = occ_good_men_house_pick }
			}
			set_province_flag = occ_good_men_house_pick
			save_event_target_as = occ_good_men_house_$n$
		}
	}
}

# Province scope. House of the Good Men (occ_houses_of_good_men): modifier plus a tax building upgrade
occ_add_good_men_house = {
	add_province_modifier = {
		name = occ_good_men_house_modifier
		duration = -1
	}
	add_or_upgrade_tax_building = yes
}

# Paratge interactions (common/government_mechanics/DST_occ_paratge.txt)
# Patronise the Troubadours: +1 Paratge per 5 prestige, at most 20, doubled with an artist
occ_troubadours_paratge_effect = {
	export_to_variable = {
		which = occ_troubadour_songs
		value = trigger_value:prestige
	}
	divide_variable = {
		which = occ_troubadour_songs
		value = 5
	}
	if = {
		limit = {
			check_variable = {
				which = occ_troubadour_songs
				value = 20
			}
		}
		set_variable = {
			which = occ_troubadour_songs
			value = 20
		}
	}
	if = {
		limit = {
			advisor = artist
		}
		multiply_variable = {
			which = occ_troubadour_songs
			value = 2
		}
	}
	while = {
		limit = {
			check_variable = {
				which = occ_troubadour_songs
				value = 1
			}
		}
		add_government_power = {
			mechanic_type = occ_paratge_mechanic
			power_type = paratge
			value = 1
		}
		subtract_variable = {
			which = occ_troubadour_songs
			value = 1
		}
	}
	set_variable = {
		which = occ_troubadour_songs
		value = 0
	}
}

# Open the Gates: +1 base production in one city, preferring centres of trade and trade buildings.
# Marks the province so repeated calls pick different ones; the interaction clears the flag afterwards
occ_open_gates_settle_effect = {
	if = {
		limit = {
			any_owned_province = {
				is_city = yes
				NOT = { has_province_flag = occ_gates_settled_flag }
				OR = {
					province_has_center_of_trade_of_level = 1
					has_building = marketplace
					has_building = trade_depot
					has_building = stock_exchange
				}
			}
		}
		random_owned_province = {
			limit = {
				is_city = yes
				NOT = { has_province_flag = occ_gates_settled_flag }
				OR = {
					province_has_center_of_trade_of_level = 1
					has_building = marketplace
					has_building = trade_depot
					has_building = stock_exchange
				}
			}
			add_base_production = 1
			set_province_flag = occ_gates_settled_flag
		}
	}
	else = {
		random_owned_province = {
			limit = {
				is_city = yes
				NOT = { has_province_flag = occ_gates_settled_flag }
			}
			add_base_production = 1
			set_province_flag = occ_gates_settled_flag
		}
	}
}

# Public Disputation, Zeal side: the most developed owned province of another faith converts (checked in steps of 20/15/10 dev)
occ_disputation_convert_effect = {
	if = {
		limit = {
			any_owned_province = {
				is_city = yes
				has_owner_religion = no
				development = 20
			}
		}
		random_owned_province = {
			limit = {
				is_city = yes
				has_owner_religion = no
				development = 20
			}
			change_religion = ROOT
		}
	}
	else_if = {
		limit = {
			any_owned_province = {
				is_city = yes
				has_owner_religion = no
				development = 15
			}
		}
		random_owned_province = {
			limit = {
				is_city = yes
				has_owner_religion = no
				development = 15
			}
			change_religion = ROOT
		}
	}
	else_if = {
		limit = {
			any_owned_province = {
				is_city = yes
				has_owner_religion = no
				development = 10
			}
		}
		random_owned_province = {
			limit = {
				is_city = yes
				has_owner_religion = no
				development = 10
			}
			change_religion = ROOT
		}
	}
	else = {
		random_owned_province = {
			limit = {
				is_city = yes
				has_owner_religion = no
			}
			change_religion = ROOT
		}
	}
}

# Council of the Good Men (occ_council_of_good_men): seat in Rome if ours and Cathar, else Toulouse, else the capital.
# Saves occ_council_seat for occ_events.13
occ_council_seat_effect = {
	if = {
		limit = {
			118 = {
				owned_by = ROOT
				religion = cathar
			}
		}
		118 = {
			save_event_target_as = occ_council_seat
			add_province_modifier = {
				name = occ_council_seat_modifier
				duration = -1
			}
		}
	}
	else_if = {
		limit = { owns = 196 }
		196 = {
			save_event_target_as = occ_council_seat
			add_province_modifier = {
				name = occ_council_seat_modifier
				duration = -1
			}
		}
	}
	else = {
		capital_scope = {
			save_event_target_as = occ_council_seat
			add_province_modifier = {
				name = occ_council_seat_modifier
				duration = -1
			}
		}
	}
}
