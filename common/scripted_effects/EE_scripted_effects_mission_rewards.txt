	custom_tooltip = BYZ_end_preview

	if = {
		limit = { can_preview_mission_of_key = { key = ADU } }
		select_current_missions = { key = ADU }

# Vanilla's update_mission_previews (on_change_tag_effect) ends every open batch when the country has none of
# vanilla's branching missions, but only clears vanilla keys. This reopens a mod preview it left without buttons.
EE_restore_branching_preview_of_key_and_batch = {
	if = {
		limit = {
			can_preview_mission_of_key_and_batch = { key = $key$ batch = $batch$ }
			NOT = { has_country_flag = has_batch_$batch$_active }
		}
		set_country_flag = has_batch_$batch$_active
		change_variable = {
			which = can_preview_missions_var
			value = 1
		}
	}
}
EE_restore_branching_previews = {
	EE_restore_branching_preview_of_key_and_batch = { key = GV batch = 1 }
	EE_restore_branching_preview_of_key_and_batch = { key = GV batch = 2 }
	EE_restore_branching_preview_of_key_and_batch = { key = GV batch = 3 }
	EE_restore_branching_preview_of_key_and_batch = { key = FAF batch = 1 }
	EE_restore_branching_preview_of_key_and_batch = { key = FAF batch = 2 }
	EE_restore_branching_preview_of_key_and_batch = { key = FAF batch = 3 }
}

end_an_incident_tree_effect = {
	if = {
		limit = {
			has_country_flag = MAM_ottoman_incident_tree
		}
		custom_tooltip = MAM_end_ottoman_incident_tooltip
		MAM_end_ottoman_incident = yes
	}
	if = {
		limit = {
			has_country_flag = GRA_incident_castile
		}
		GRA_end_incident = yes
	}
}
